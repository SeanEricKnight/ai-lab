from pathlib import Path
import argparse
import platform
import shutil
import subprocess
import sys

import yaml

VERSION = "0.1.0"


def repo_root() -> Path:
    return Path(__file__).resolve().parents[2]


def objects_dir() -> Path:
    return repo_root() / "objects"


def run_command(command: list[str]) -> tuple[bool, str]:
    try:
        result = subprocess.run(
            command,
            capture_output=True,
            text=True,
            check=False,
        )
        output = (result.stdout or result.stderr).strip()
        return result.returncode == 0, output
    except FileNotFoundError:
        return False, "not found"


def load_asset(asset_id: str):
    path = objects_dir() / "assets" / asset_id / "asset.yaml"
    if not path.exists():
        raise FileNotFoundError(f"Asset not found: {asset_id}")
    return yaml.safe_load(path.read_text())


def list_assets():
    root = objects_dir() / "assets"
    rows = []
    for path in sorted(root.glob("*/asset.yaml")):
        rows.append(yaml.safe_load(path.read_text()))
    return rows


def status_line(name: str, ok: bool, detail: str = ""):
    mark = "OK" if ok else "MISSING"
    if detail:
        print(f"{mark:<8} {name:<20} {detail}")
    else:
        print(f"{mark:<8} {name}")


def cmd_version(_args):
    print(f"AI Lab {VERSION}")
    print(f"Repository: {repo_root()}")


def cmd_asset_list(_args):
    print("Assets")
    print("------")
    for asset in list_assets():
        print(f"{asset.get('id', ''):<8} {asset.get('hostname', ''):<12} {asset.get('role', '')}")


def cmd_asset_show(args):
    try:
        asset = load_asset(args.asset_id)
    except FileNotFoundError as exc:
        print(str(exc), file=sys.stderr)
        return 1

    print(f"{asset.get('id')} - {asset.get('hostname')}")
    print("-" * 40)
    for key, value in asset.items():
        if isinstance(value, dict):
            print(f"{key}:")
            for sub_key, sub_value in value.items():
                print(f"  {sub_key}: {sub_value}")
        else:
            print(f"{key}: {value}")
    return 0


def cmd_workstation_audit(_args):
    print("PowerSpec Workstation Audit")
    print("===========================")
    print(f"Repository: {repo_root()}")
    print(f"Platform:   {platform.platform()}")
    print()

    checks = [
        ("PowerShell", ["pwsh", "--version"]),
        ("Git", ["git", "--version"]),
        ("VS Code", ["code", "--version"]),
        ("Python", ["python", "--version"]),
        ("Conda", ["conda", "--version"]),
        ("Mamba", ["mamba", "--version"]),
        ("Ollama", ["ollama", "--version"]),
        ("Docker", ["docker", "--version"]),
        ("Docker Compose", ["docker", "compose", "version"]),
        ("WSL", ["wsl", "--status"]),
    ]

    for name, command in checks:
        ok, output = run_command(command)
        first_line = output.splitlines()[0] if output else ""
        status_line(name, ok, first_line)

    print()
    git_ok, git_status = run_command(["git", "status", "--short"])
    clean = git_ok and git_status == ""
    status_line("Git working tree", clean, "clean" if clean else git_status)

    return 0


def main():
    parser = argparse.ArgumentParser(prog="labctl")
    sub = parser.add_subparsers(dest="command")

    p_version = sub.add_parser("version")
    p_version.set_defaults(func=cmd_version)

    p_asset = sub.add_parser("asset")
    asset_sub = p_asset.add_subparsers(dest="asset_command")

    p_asset_list = asset_sub.add_parser("list")
    p_asset_list.set_defaults(func=cmd_asset_list)

    p_asset_show = asset_sub.add_parser("show")
    p_asset_show.add_argument("asset_id")
    p_asset_show.set_defaults(func=cmd_asset_show)

    p_workstation = sub.add_parser("workstation")
    workstation_sub = p_workstation.add_subparsers(dest="workstation_command")

    p_workstation_audit = workstation_sub.add_parser("audit")
    p_workstation_audit.set_defaults(func=cmd_workstation_audit)

    args = parser.parse_args()

    if not hasattr(args, "func"):
        parser.print_help()
        return 1

    result = args.func(args)
    return 0 if result is None else result


if __name__ == "__main__":
    raise SystemExit(main())

I have the coding share but its on the cache - maybe i should put it on the array?

For coding, I would not make it array-only.

Best choice for your lab:

Primary storage: cache
Secondary storage: array
Mover action: cache -> array

Why: active code benefits from SSD speed, but the array gives you long-term capacity/protection. Unraid’s current share model uses Primary storage for new files, Secondary storage for overflow, and Mover to transfer files between cache and array.

My recommendation:
Share	Best setting
appdata	cache only
system	cache only
coding	cache primary, array secondary
work	cache primary, array secondary
documents	array primary
media	array primary
downloads	dedicated download SSDs later

For coding, this gives you fast local-like performance from PowerSpec while still allowing Mover to push older files to the array. If you want active repos to always stay fast, then make coding cache only and back it up separately to the array/GitHub/JARVIS backup.
